cmake_minimum_required(VERSION 3.20)
project(tinium_bitchord_prototype VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  set(host_default OFF)
else()
  set(host_default ON)
endif()
option(TINIUM_BUILD_TOOLS "Build host CLI/reference/tests" ${host_default})
option(TINIUM_SANITIZERS "Enable host address/undefined sanitizers" OFF)
option(TINIUM_BUILD_SHARED "Build standalone shared library (Android JNI links the static core)" ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  if(TINIUM_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitchord/tempo_analysis.cpp" tempo_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitchord/resampler.cpp" resample_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitchord/resampler.h" resample_header_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitchord/audio_analysis.h" tempo_header_hash)
if(NOT tempo_header_hash STREQUAL "679ab56bd8d6e632c88cbfe015dbc893809deb606a97ccaebecf3c73e306b192" OR NOT tempo_hash STREQUAL "f6cde3e8a87fb1f42e356649fffe2a87b36f857be28d360b326c14d9c478bf2f" OR
   NOT resample_hash STREQUAL "f516a5683d77537c73f628a4e45f44842ef4a5b3c6d9938d2014ac05e83884a0" OR
   NOT resample_header_hash STREQUAL "93ae72db79f64efb5c8e0bb3296a565ae7286185482de781134bccb621ce149e")
  message(FATAL_ERROR "Pinned BitChord source hash mismatch")
endif()
include_directories(include src vendor/bitchord)
include(cmake/ControlledSources.cmake)
add_library(native_objects OBJECT src/analyzer.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/controlled/tempo_analysis.cpp"
  "${CMAKE_CURRENT_BINARY_DIR}/controlled/resampler.cpp")
add_library(tinium_native STATIC $<TARGET_OBJECTS:native_objects>)
target_include_directories(tinium_native PUBLIC include)
if(TINIUM_BUILD_SHARED)
  add_library(tinium_native_shared SHARED $<TARGET_OBJECTS:native_objects>)
  if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(tinium_native_shared PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
endif()
if(TINIUM_BUILD_TOOLS)
  find_package(Threads REQUIRED)
  add_library(upstream_reference STATIC vendor/bitchord/tempo_analysis.cpp vendor/bitchord/resampler.cpp)
  add_executable(analyzer_cli src/cli.cpp)
  target_link_libraries(analyzer_cli PRIVATE tinium_native)
  add_executable(reference_cli src/cli.cpp src/analyzer.cpp)
  target_compile_definitions(reference_cli PRIVATE TINIUM_UNMODIFIED_REFERENCE=1)
  target_link_libraries(reference_cli PRIVATE upstream_reference)
  enable_testing()
  add_executable(native_tests tests/native_tests.cpp)
  target_link_libraries(native_tests PRIVATE tinium_native upstream_reference Threads::Threads)
  foreach(suite input output parity rhythm uncertainty resampling cancellation concurrency)
    add_test(NAME ${suite} COMMAND native_tests ${suite})
    set_tests_properties(${suite} PROPERTIES TIMEOUT 120)
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py" "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME provenance COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_provenance.py")
    set_tests_properties(tools provenance PROPERTIES TIMEOUT 120)
  endif()
endif()
