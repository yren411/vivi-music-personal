cmake_minimum_required(VERSION 3.22.1)
project(tinium_automix_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(TINIUM_BUILD_TOOLS OFF CACHE BOOL "No benchmark/reference code in APK" FORCE)
set(TINIUM_BUILD_SHARED OFF CACHE BOOL "Only package JNI library" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../experiments/bitchord-native-analysis" "${CMAKE_CURRENT_BINARY_DIR}/bitchord")
set_target_properties(native_objects PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(tinium_automix SHARED native_analysis_jni.cpp)
set_target_properties(tinium_automix PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_libraries(tinium_automix PRIVATE tinium_native)
if(NOT ANDROID)
  # Allows genuine JNI tests on a host JVM; does not substitute for Android runtime tests.
  find_package(JNI REQUIRED COMPONENTS JVM)
  target_include_directories(tinium_automix PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(tinium_automix PRIVATE "-Wl,-z,max-page-size=16384")
endif()
